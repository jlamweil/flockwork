## 2. The mm2 serve sbatch (staged body — house lines per fleet law)

```bash
#!/bin/bash
#SBATCH --job-name=mm2-vllm-tp1
#SBATCH --output=mm2-vllm-tp1-%j.out
#SBATCH --error=mm2-vllm-tp1-%j.err
#SBATCH --time=06:00:00            # the teardown — no --requeue (freeze §6)
#SBATCH --nodes=1
#SBATCH --gres=gpu:1
#SBATCH --mem=64G
#SBATCH --constraint=gpuh100
#SBATCH --partition=publicgpu
<slurm-account>
#SBATCH --qos=defaultqos
#SBATCH --exclude=hpc-n968,hpc-n977   # owner-banned nodes
set -uo pipefail
VLLM_PYTHON="${VLLM_PYTHON:-$HOME/.conda/envs/autogluon-env-v125/bin/python}"
export CUDA_PATH="${CUDA_PATH:-/usr/local/cuda/13.0}"
module load cuda/13.0 2>/dev/null || true
export HF_HOME="$HOME/hf_cache"      # weights already local (§0 download)
export HF_HUB_OFFLINE=1
exec "$VLLM_PYTHON" -m vllm.entrypoints.openai.api_server \
  --model Qwen/Qwen2.5-Coder-7B-Instruct \
  --served-model-name Qwen2.5-Coder-7B-Instruct \
  --port serve-port-b --tensor-parallel-size 1 \
  --max-model-len 32768 --gpu-memory-utilization 0.90
```

(If the v125 env's vLLM wants `python -m vllm serve` instead, that form
is equivalent — the pinned parts are: served name, port serve-port-b, tp1,
32768, offline-from-`$HOME/hf_cache`.)
